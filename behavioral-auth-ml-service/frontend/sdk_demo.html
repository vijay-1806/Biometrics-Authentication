3. How Any Web App Integrates This in 15 Lines of Code

<!-- 1. Include the SDK snippet in your HTML <head> -->
<script src="http://localhost:8000/sdk.js"></script>

<script>
  // 2. Initialize BioAuth with your tenant API key and candidate ID
  const bio = new BioAuth({
    apiKey: 'bio_live_your_tenant_key',
    userId: 'student_12345'
  });

  // 3. Launch calibration when needed (Zero UI work required)
  bio.openEnrollmentModal({
    targetSamples: 20,
    onComplete: (res) => console.log('Biometric model calibrated & ready!', res)
  });

  // 4. Start real-time proctoring during assessments
  bio.startSession({
    examId: 'cs101_midterm',
    onScore: (score) => console.log('Live Trust Score:', score.trust_score),
    onViolation: (v) => alert('Proctor Violation Detected: ' + v.type)
  });
</script>
